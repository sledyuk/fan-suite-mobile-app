```ts
// src/storage/KeyValueStorage.ts
export interface KeyValueStorage { get(key: string): string | null; set(key: string, value: string): void; remove(key: string): void; keys(): string[]; clear(): void }
export class NamespacedKV implements KeyValueStorage {
  constructor(private inner: KeyValueStorage, private prefix: string) {}
  get(k: string) { return this.inner.get(this.prefix + k); }
  set(k: string, v: string) { this.inner.set(this.prefix + k, v); }
  remove(k: string) { this.inner.remove(this.prefix + k); }
  keys() { return this.inner.keys().filter(k => k.startsWith(this.prefix)).map(k => k.slice(this.prefix.length)); }
  clear() { for (const k of this.keys()) this.remove(k); }
}
```

```ts
// MockChatServer.ts
import type { ChatApi } from '../api/ChatApi'; import { SendError, type ClientId, type ServerId, type ServerMessage } from '../api/types';
import type { KeyValueStorage } from '@/storage/KeyValueStorage'; import type { Faults } from './faults'; import { generateHistory } from './historyGenerator';
interface Persisted { messages: ServerMessage[]; acceptedByClientId: Record<ClientId, ServerId>; nextSeq: number }
const KEY = 'mock-chat-server.v1'; const sleep = (ms: number) => new Promise(r => setTimeout(r, ms));
export class MockChatServer implements ChatApi {
  protected state: Persisted;
  constructor(protected storage: KeyValueStorage, protected faults: () => Faults, protected opts: { seedCount?: number } = {}) {
    const raw = storage.get(KEY); this.state = raw ? JSON.parse(raw) : this.seed(); if (!raw) this.persist();
  }
  private seed(): Persisted { const m = generateHistory(this.opts.seedCount ?? 50_000); return { messages: m, acceptedByClientId: {}, nextSeq: m.length + 1 }; }
  protected persist() { this.storage.set(KEY, JSON.stringify(this.state)); }
  protected async gate() { const f = this.faults(); if (f.latencyMs) await sleep(f.latencyMs); if (f.offline) throw new SendError('NETWORK', true, 'You are offline'); }
  protected accept(input: { clientId: ClientId; text: string; createdAt: number }): ServerMessage {
    const msg: ServerMessage = { id: `m_${this.state.nextSeq}`, clientId: input.clientId, seq: this.state.nextSeq++, authorId: 'fan', text: input.text, createdAt: Date.now(), kind: 'text' };
    this.state.messages.push(msg); this.state.acceptedByClientId[input.clientId] = msg.id; this.persist(); return msg;
  }
  protected maybeDropResponse() { const f = this.faults(); if (f.dropNextResponse) { f.dropNextResponse = false; throw new SendError('NETWORK', true, 'Response lost'); } }
  protected maybeFail() { const f = this.faults(); if (!f.failNextSend) return; const code = f.failNextSend; f.failNextSend = null;
    if (code === 'RATE_LIMITED') throw new SendError(code, true, 'Too many messages, retry in a moment');
    if (code === 'PAYMENT_REQUIRED') throw new SendError(code, false, 'Subscribe to send messages');
    throw new SendError('BLOCKED', false, "You can't message this creator"); }
  async send(input: { clientId: ClientId; text: string; createdAt: number }) {
    await this.gate(); this.maybeFail();
    const existing = this.state.acceptedByClientId[input.clientId];
    if (existing) return this.state.messages.find(m => m.id === existing)!;        // idempotent retry
    const msg = this.accept(input); this.maybeDropResponse(); return msg;
  }
  async sync(sinceSeq: number) { await this.gate(); return this.state.messages.filter(m => m.seq > sinceSeq); }
  async getPage(beforeSeq: number | null, limit: number) {
    await this.gate(); const all = this.state.messages;
    const endIdx = beforeSeq === null ? all.length : Math.max(0, all.findIndex(m => m.seq >= beforeSeq)); const end = beforeSeq !== null && endIdx === 0 && all[0]?.seq < beforeSeq ? all.length : endIdx;
    const start = Math.max(0, end - limit); return { messages: all.slice(start, end), hasMore: start > 0 };
  }
  injectIncoming(texts: string[]) { return texts.map(text => { const m: ServerMessage = { id: `m_${this.state.nextSeq}`, seq: this.state.nextSeq++, authorId: 'creator', text, createdAt: Date.now(), kind: 'text' }; this.state.messages.push(m); this.persist(); return m; }); }
  reset() { this.state = this.seed(); this.persist(); }
  messageCount() { return this.state.messages.length; } allMessages() { return this.state.messages; }
}
```

```ts
// persist.ts
import { reaction, runInAction } from 'mobx'; import type { KeyValueStorage } from '@/storage/KeyValueStorage';
export function persistSlice<T>(storage: KeyValueStorage, key: string, read: () => T, apply: (v: T) => void) {
  const raw = storage.get(key); if (raw) { try { runInAction(() => apply(JSON.parse(raw) as T)); } catch { storage.remove(key); } }
  return reaction(() => JSON.stringify(read()), (json) => storage.set(key, json));
}
```

```ts
export class OutboxStore {
  items: OutboxItem[] = []; dispose: () => void;
  constructor(storage: KeyValueStorage) { makeAutoObservable(this, { dispose: false });
    this.dispose = persistSlice(storage, 'outbox.v1', () => this.items, v => { this.items = v.map(i => i.status === 'sending' ? { ...i, status: 'pending' } : i); }); }
  private find(id: string) { const it = this.items.find(i => i.clientId === id); if (!it) throw new Error(`outbox: unknown ${id}`); return it; }
  enqueue(text: string, clientId = Crypto.randomUUID()) { const item: OutboxItem = { clientId, text, createdAt: Date.now(), status: 'pending', attempts: 0 }; this.items.push(item); return item; }
  markSending(id: string) { const it = this.find(id); it.status = 'sending'; it.attempts += 1; }
```

```ts
import { reaction, runInAction } from 'mobx'; import type { RootStore } from '@/stores/RootStore'; import type { ChatApi } from '@/services/api/ChatApi'; import { SendError } from '@/services/api/types';
const BACKOFF_MS = [500, 1500, 4000];
export async function drainOnce(root: RootStore, api: ChatApi): Promise<void> {
  const { outbox, chat, connectivity } = root;
  if (!connectivity.online || outbox.sendingOne) return;
  const next = outbox.pending[0]; if (!next) return;
  runInAction(() => outbox.markSending(next.clientId));
  try { const msg = await api.send({ clientId: next.clientId, text: next.text, createdAt: next.createdAt }); runInAction(() => { chat.upsert([msg]); outbox.remove(next.clientId); }); }
  catch (e) {
    const err = e instanceof SendError ? e : new SendError('NETWORK', true, 'Network error');
    runInAction(() => {
      if (err.code === 'NETWORK') { outbox.markPending(next.clientId); const attempts = outbox.items.find(i => i.clientId === next.clientId)?.attempts ?? 1;
        if (attempts >= BACKOFF_MS.length) outbox.markFailed(next.clientId, { code: 'NETWORK', recoverable: true, message: 'Could not reach the server' });

    });

}
```
