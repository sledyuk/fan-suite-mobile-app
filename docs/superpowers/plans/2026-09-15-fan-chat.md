```ts
export function mulberry32(seed: number) { return () => { let t = (seed += 0x6d2b79f5); t = Math.imul(t ^ (t >>> 15), t | 1); t ^= t + Math.imul(t ^ (t >>> 7), t | 61); return ((t ^ (t >>> 14)) >>> 0) / 4294967296; }; }
const SNIPPETS = ['hey!', 'did you see the new post?', 'love it 🔥', 'thanks so much', 'when is the next stream?', 'tonight at 8', 'sent you a little gift', 'haha yes', 'ok see you', 'that clip was amazing', 'can you sign my print?', 'sure, DM me the address'];
export function generateHistory(count: number, seed = 42, startAt = Date.UTC(2026, 0, 1)): ServerMessage[] {
  const rnd = mulberry32(seed); const out: ServerMessage[] = []; let t = startAt;
  for (let i = 0; i < count; i++) {

  }
  return out;
}
```

```ts
// src/services/mock/MockChatServer.ts
import type { ChatApi } from '../api/ChatApi';
import { SendError, type ClientId, type ServerId, type ServerMessage } from '../api/types';
import type { Storage } from '@/storage/mmkv';
import type { Faults } from './faults';
import { generateHistory } from './historyGenerator';

interface Persisted { messages: ServerMessage[]; acceptedByClientId: Record<ClientId, ServerId>; nextSeq: number }
const KEY = 'mock-chat-server.v1';
const sleep = (ms: number) => new Promise(r => setTimeout(r, ms));

export class MockChatServer implements ChatApi {
  protected state: Persisted;
  constructor(protected storage: Storage, protected faults: () => Faults, opts: { seedCount?: number } = {}) {
    const raw = storage.getString(KEY);
    if (raw) this.state = JSON.parse(raw);
    else { const seed = generateHistory(opts.seedCount ?? 50_000); this.state = { messages: seed, acceptedByClientId: {}, nextSeq: seed.length + 1 }; this.persist(); }
  }
  protected persist() { this.storage.set(KEY, JSON.stringify(this.state)); }
  protected async gate() {
    const f = this.faults(); if (f.latencyMs) await sleep(f.latencyMs);
    if (f.offline) throw new SendError('NETWORK', true, 'You are offline');
  }
  protected accept(input: { clientId: ClientId; text: string; createdAt: number }): ServerMessage {
    const msg: ServerMessage = { id: `m_${this.state.nextSeq}`, clientId: input.clientId, seq: this.state.nextSeq++, authorId: 'fan', text: input.text, createdAt: Date.now(), kind: 'text' };
    this.state.messages.push(msg); this.state.acceptedByClientId[input.clientId] = msg.id; this.persist(); return msg;
  }
  protected maybeDropResponse() {
    const f = this.faults(); if (f.dropNextResponse) { f.dropNextResponse = false; throw new SendError('NETWORK', true, 'Response lost'); }
  }
  protected maybeFail() {
    const f = this.faults(); if (!f.failNextSend) return; const code = f.failNextSend; f.failNextSend = null;
    if (code === 'RATE_LIMITED') throw new SendError(code, true, 'Too many messages, retry in a moment');
    if (code === 'PAYMENT_REQUIRED') throw new SendError(code, false, 'Subscribe to send messages');
    throw new SendError('BLOCKED', false, "You can't message this creator");
  }
  async send(input: { clientId: ClientId; text: string; createdAt: number }): Promise<ServerMessage> {
    await this.gate(); this.maybeFail();
    const existingId = this.state.acceptedByClientId[input.clientId];
    if (existingId) return this.state.messages.find(m => m.id === existingId)!;   // idempotent retry
    const msg = this.accept(input); this.maybeDropResponse(); return msg;
  }
  async sync(sinceSeq: number) { await this.gate(); return this.state.messages.filter(m => m.seq > sinceSeq); }
  async getPage(beforeSeq: number | null, limit: number) {
    await this.gate();
    const all = this.state.messages; const end = beforeSeq === null ? all.length : all.findIndex(m => m.seq >= beforeSeq);
    const endIdx = end === -1 ? all.length : end; const start = Math.max(0, endIdx - limit);
    return { messages: all.slice(start, endIdx), hasMore: start > 0 };
  }
  injectIncoming(texts: string[]) { return texts.map(text => { const m: ServerMessage = { id: `m_${this.state.nextSeq}`, seq: this.state.nextSeq++, authorId: 'creator', text, createdAt: Date.now(), kind: 'text' }; this.state.messages.push(m); this.persist(); return m; }); }
  reset() { this.storage.remove(KEY); const seed = generateHistory(50_000); this.state = { messages: seed, acceptedByClientId: {}, nextSeq: seed.length + 1 }; this.persist(); }
  messageCount() { return this.state.messages.length; }
  allMessages() { return this.state.messages; }
}
```

```ts
import { createStore } from 'zustand/vanilla'; import { persist, createJSONStorage } from 'zustand/middleware';
import * as Crypto from 'expo-crypto';
import type { OutboxItem } from '@/services/api/types'; import { makeZustandStorage } from '@/storage/zustandStorage'; import type { Storage } from '@/storage/mmkv';
export interface OutboxState { items: OutboxItem[]; enqueue: (text: string, clientId?: string) => OutboxItem; markSending: (id: string) => void; markPending: (id: string) => void; markFailed: (id: string, error: NonNullable<OutboxItem['error']>) => void; remove: (id: string) => void; retry: (id: string) => void; clear: () => void }
const patch = (items: OutboxItem[], id: string, p: Partial<OutboxItem>) => items.map(i => i.clientId === id ? { ...i, ...p } : i);
export const createOutboxStore = (storage: Storage) => createStore<OutboxState>()(persist((set, get) => ({
  items: [],
  enqueue: (text, clientId = Crypto.randomUUID()) => { const item: OutboxItem = { clientId, text, createdAt: Date.now(), status: 'pending', attempts: 0 }; set({ items: [...get().items, item] }); return item; },
  markSending: (id) => set(s => ({ items: patch(s.items, id, { status: 'sending', attempts: (s.items.find(i => i.clientId === id)?.attempts ?? 0) + 1 }) })),
  markPending: (id) => set(s => ({ items: patch(s.items, id, { status: 'pending', error: undefined }) })),
  markFailed: (id, error) => set(s => ({ items: patch(s.items, id, { status: 'failed', error }) })),
  remove: (id) => set(s => ({ items: s.items.filter(i => i.clientId !== id) })),
  retry: (id) => set(s => ({ items: patch(s.items, id, { status: 'pending', error: undefined, attempts: 0 }) })),
  clear: () => set({ items: [] }),
}), {
  name: 'outbox.v1', storage: createJSONStorage(() => makeZustandStorage(storage)),
  partialize: (s) => ({ items: s.items }),
  merge: (persisted, current) => ({ ...current, items: ((persisted as { items?: OutboxItem[] })?.items ?? []).map(i => i.status === 'sending' ? { ...i, status: 'pending' as const } : i) }),
}));
```

```ts
// src/stores/chatStore.ts
import { createStore } from 'zustand/vanilla'; import type { ServerId, ServerMessage } from '@/services/api/types';
export interface ChatState { byId: Record<ServerId, ServerMessage>; orderedIds: ServerId[]; lastSeq: number; oldestLoadedSeq: number | null; hasMore: boolean; loadingOlder: boolean; upsert: (msgs: ServerMessage[]) => void; setPage: (msgs: ServerMessage[], hasMore: boolean) => void; setLoadingOlder: (b: boolean) => void; clear: () => void }
export const createChatStore = () => createStore<ChatState>()((set, get) => ({
  byId: {}, orderedIds: [], lastSeq: 0, oldestLoadedSeq: null, hasMore: true, loadingOlder: false,
  upsert: (msgs) => { const { byId, orderedIds } = get(); let changed = false; const next = { ...byId };
    for (const m of msgs) { if (!byId[m.id]) changed = true; next[m.id] = m; }
    if (!changed) return;
```

```ts
// src/workers/outboxDrainer.ts
import type { Stores } from '@/stores/createStores'; import type { ChatApi } from '@/services/api/ChatApi'; import { SendError } from '@/services/api/types';
const BACKOFF = [500, 1500, 4000];
export async function drainOnce(stores: Stores, api: ChatApi): Promise<void> {
  const { outbox, chat, connectivity } = stores;
  if (!connectivity.getState().online) return;
  if (outbox.getState().items.some(i => i.status === 'sending')) return;
  const next = outbox.getState().items.find(i => i.status === 'pending'); if (!next) return;
  outbox.getState().markSending(next.clientId);
  try {
    const msg = await api.send({ clientId: next.clientId, text: next.text, createdAt: next.createdAt });
    chat.getState().upsert([msg]); outbox.getState().remove(next.clientId);
  } catch (e) {
    const err = e instanceof SendError ? e : new SendError('NETWORK', true, 'Network error');
    if (err.code === 'NETWORK') { outbox.getState().markPending(next.clientId);
      const attempts = outbox.getState().items.find(i => i.clientId === next.clientId)?.attempts ?? 1;
      if (attempts < BACKOFF.length && connectivity.getState().online) setTimeout(() => void drainOnce(stores, api), BACKOFF[attempts - 1]);
      else if (attempts >= BACKOFF.length) outbox.getState().markFailed(next.clientId, { code: 'NETWORK', recoverable: true, message: 'Could not reach the server' });
      return; }
    if (err.recoverable) outbox.getState().markFailed(next.clientId, { code: err.code, recoverable: true, message: err.message });
    else outbox.getState().markFailed(next.clientId, { code: err.code, recoverable: false, message: err.message });
  }
  void drainOnce(stores, api); // continue with the next pending item
}
export function startOutboxDrainer(stores: Stores, api: ChatApi) {
  const unsub = stores.outbox.subscribe((s, prev) => { if (s.items !== prev.items) void drainOnce(stores, api); });
  void drainOnce(stores, api); return unsub;
}
```
